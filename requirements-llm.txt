# Optional: the *alternative* way AURA can run a model.
#
# AURA has two local backends. The default one is built in - it downloads the
# official prebuilt llama.cpp `llama-server` and runs it as a child process
# (see aura/llama_server.py), so there is nothing to install here.
#
# This file is for the other one: llama.cpp as a Python extension, which loads
# the .gguf inside AURA's own process. It is only useful if you specifically
# want that (no child process, or a llama-cpp-python build with hardware
# acceleration), and it is why the package is not in requirements.txt.
#
#     pip install -r requirements-llm.txt
#     # then set Settings -> "How answers are written" to `llamacpp`
#
# Wheels for CPU builds:
#     pip install llama-cpp-python --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cpu
#
# Nothing else in AURA depends on this package, which keeps the shipped build
# small and dependency-free.

llama-cpp-python>=0.2.79
